```text
  prompts.ts                   ← 上一份 generate_confucius/
  sentenceOrganization.ts
  evaluate_question_confucius.ts
  qwenJson.ts                  ← 本包根上
  corpus.ts …                  ← 本包 select_confucius/
  moodDetect.ts …              ← 本包 generate_confucius/
src/utils/confuciusExpression.ts  ← 本包 utils/
src/data/confucius/            ← 上一份 zip 的 JSON（统一文件名）
```
